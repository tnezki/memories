  4 versions x 2 pages = 8 portrait pages.

JAVASCRIPT — EXACT MECHANISM:

  function clearExitPrintTarget(){
    document.querySelectorAll('.exit-print.print-active').forEach(function(el){
      el.classList.remove('print-active');
    });
  }

  function printExitTicket(ticketId){
    clearExitPrintTarget();
    var target=document.querySelector('.exit-print[data-source-ticket="'+ticketId+'"]');
    if(!target){return;}
    target.classList.add('print-active');
    window.setTimeout(function(){window.print();},30);
  }

  window.addEventListener('afterprint',clearExitPrintTarget);
